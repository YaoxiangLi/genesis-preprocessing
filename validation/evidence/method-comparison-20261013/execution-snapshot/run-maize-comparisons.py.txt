"""Run the remaining approved maize panel after verified preparation and other cohorts.

This local driver preserves the frozen scientific recipe, serial execution,
aggregate wall/storage limits and a receipt for each attempted command.
"""
from __future__ import annotations

import datetime
import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import time

ROOT = Path.cwd()
BASE = ROOT / 'results/method-benchmark-20261013'
RECIPE = ROOT / 'benchmarks/experiments/plant-methods-20261013'
PYTHON = str(ROOT / 'genesis_tools/.venv/bin/python')
PLOT = str(ROOT / 'benchmarks/environment/.pixi/envs/default/bin/python')
BENCH = [PYTHON, '-m', 'genesis_tools.benchmark']
ENV = dict(os.environ, PYTHONPATH=str(ROOT / 'genesis_tools/src'), PYTHONUNBUFFERED='1')
DEADLINE = datetime.datetime(2026, 10, 15, 6, 12, 46, tzinfo=datetime.UTC).timestamp()
EXPECTED = json.loads((BASE / 'synthetic-dap-v5-results/experiment.json').read_text())['source']['recipe_sha256']
RECEIPTS: list[dict[str, object]] = []
PATHS = [BASE, Path('/var/tmp/genesis-public-workflow-20261013'), Path('/var/tmp/genesis-whole-workflow-20261013'), Path('/var/tmp/genesis-method-validation-20261013'), Path('/var/tmp/genesis-installed-package-fix-20261013')]


def stamp() -> str:
    return datetime.datetime.now(datetime.UTC).isoformat()


def frozen() -> None:
    for name, digest in EXPECTED.items():
        if hashlib.sha256((ROOT / 'genesis_tools/src/genesis_tools' / name).read_bytes()).hexdigest() != digest:
            raise RuntimeError('Frozen comparison recipe changed: ' + name)


def remaining() -> None:
    if time.time() > DEADLINE - 21600:
        raise RuntimeError('Less than six hours remain in the aggregate study budget')


def storage() -> int:
    totals = {}
    for root in PATHS:
        size = 0
        for directory, folders, names in os.walk(root):
            folders[:] = [d for d in folders if d not in {'.git', '.pixi', '.venv', '__pycache__'} and (Path(directory) / d) != BASE / 'sources']
            for name in names:
                path = Path(directory) / name
                try:
                    if not path.is_symlink():
                        size += path.stat().st_size
                except FileNotFoundError:
                    pass
        totals[str(root)] = size
    total = sum(totals.values())
    with (BASE / 'maize-storage-observations.jsonl').open('a') as stream:
        stream.write(json.dumps({'utc': stamp(), 'bytes': totals, 'total': total, 'scope': 'apparent bytes; verified/original acquisition inputs and installed software excluded; sampled, not exact peak'}) + '\n')
    return total


def run(argv: list[str]) -> None:
    frozen()
    remaining()
    if storage() > 78000000000:
        raise RuntimeError('Insufficient aggregate temporary-work budget')
    start = time.monotonic()
    record: dict[str, object] = {'argv': argv, 'utc': stamp()}
    print('COMMAND', record['utc'], repr(argv), flush=True)
    process = subprocess.Popen(argv, env=ENV)
    last_check = 0.0
    while process.poll() is None:
        if time.monotonic() - last_check >= 15:
            last_check = time.monotonic()
            if storage() > 78000000000 or time.time() > DEADLINE:
                record['stop_reason'] = 'Aggregate temporary-work or wall-time guard'
                process.send_signal(signal.SIGINT)
                break
        time.sleep(1)
    code = process.wait()
    record.update(exit=code, seconds=time.monotonic() - start)
    RECEIPTS.append(record)
    (BASE / 'maize-comparisons-commands.json').write_text(json.dumps(RECEIPTS, indent=2) + '\n')
    if code or 'stop_reason' in record:
        raise RuntimeError('Command failed; work retained for diagnosis')


print('WAITING for full verified maize preparation and completed DAP/Arabidopsis depths', stamp(), flush=True)
while True:
    remaining()
    preparation = BASE / 'maize-preparation-command.json'
    prepared = preparation.exists() and json.loads(preparation.read_text())['exit'] == 0
    if preparation.exists() and not prepared:
        raise RuntimeError('Maize preparation failed; inspect its retained receipt')
    completed = True
    for cohort in ('dap-se', 'dap-pe', 'atac-arabidopsis'):
        path = ROOT / f'validation/evidence/method-comparison-20261013/{cohort}-5m/manifest.json'
        completed = completed and path.exists() and json.loads(path.read_text())['prune_status'] == 'PRUNED'
    if prepared and completed:
        break
    time.sleep(30)

for depth, repeats in ((1000000, 3), (5000000, 1)):
    label = f'{depth // 1000000}m'
    for group in ('mapping', 'processing'):
        spec = BASE / 'atac-maize' / f'comparison-{label}-{group}-v5.json'
        result = BASE / f'atac-maize-{label}-{group}-v5-results'
        run([PYTHON, str(RECIPE / 'compile_matrix.py'), str(BASE / 'atac-maize' / f'atac-maize-depth-{depth}.json'), '--out', str(spec), '--group', group, '--repetitions', str(repeats), '--cpus', '14'])
        available = min(70000000000, 80000000000 - storage() - 2000000000)
        if available < 32000000000:
            raise RuntimeError('Insufficient temporary work for maize; preserve existing evidence')
        specification = json.loads(spec.read_text())
        specification['limits']['disk_bytes'] = available
        spec.write_text(json.dumps(specification, indent=2) + '\n')
        run(BENCH + ['run', str(spec), '--out', str(result)])
        run(BENCH + ['report', str(result), '--plot-python', PLOT])
        if group == 'mapping':
            run(['taskset', '-c', '14', PLOT, str(RECIPE / 'plot_profiles.py'), str(result), '--method', 'baseline', '--method', 'bwa', '--method', 'bowtie2', '--out', str(ROOT / f'validation/evidence/method-comparison-20261013/atac-maize-{label}-{group}-profiles')])
        else:
            run(['taskset', '-c', '14', PLOT, str(RECIPE / 'plot_profiles.py'), str(result), '--method', 'baseline', '--method', 'cutadapt', '--method', 'fastp', '--out', str(ROOT / f'validation/evidence/method-comparison-20261013/atac-maize-{label}-{group}-profiles')])
        run(['taskset', '-c', '14', PYTHON, str(RECIPE / 'archive.py'), str(result), '--out', str(ROOT / f'validation/evidence/method-comparison-20261013/atac-maize-{label}-{group}'), '--prune-work'])
print('COMPLETE maize panel', stamp(), flush=True)
