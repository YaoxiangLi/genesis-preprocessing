"""Prepare the maize cohort only after both complete pairs pass publisher checks."""
from pathlib import Path
import datetime,json,os,subprocess,time
root=Path.cwd();base=root/'results/method-benchmark-20261013'
os.sched_setaffinity(0,{14,15})
names=[f'{run}_{mate}.fastq.gz' for run in ['SRR27443454','SRR27443453'] for mate in [1,2]]
deadline=datetime.datetime(2026,10,15,0,12,46,tzinfo=datetime.UTC).timestamp()
while True:
 state=json.loads((base/'sources/acquisition-state.json').read_text())['assets']
 if all(state.get(name,{}).get('verified') for name in names):break
 if time.time()>deadline:raise RuntimeError('Maize acquisition not ready within preparation budget')
 time.sleep(30)
argv=[str(root/'genesis_tools/.venv/bin/python'),str(root/'benchmarks/experiments/plant-methods-20261013/prepare.py'),'--sources',str(base/'sources'),'--out',str(base/'atac-maize'),'--cohort','atac-maize','--depth','1000000','--depth','5000000']
started=time.monotonic();utc=datetime.datetime.now(datetime.UTC).isoformat();print('COMMAND',utc,repr(argv),flush=True)
r=subprocess.run(argv,env=dict(os.environ,PYTHONPATH=str(root/'genesis_tools/src'),PYTHONUNBUFFERED='1'))
(base/'maize-preparation-command.json').write_text(json.dumps({'argv':argv,'utc':utc,'seconds':time.monotonic()-started,'exit':r.returncode},indent=2)+'\n')
raise SystemExit(r.returncode)
