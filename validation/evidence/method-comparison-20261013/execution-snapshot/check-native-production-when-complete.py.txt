from pathlib import Path
import datetime,json,os,subprocess,time
root=Path.cwd();base=root/'results/method-benchmark-20261013'
source=base/'atac-arabidopsis-1m-v5-results/results.json'
deadline=datetime.datetime(2026,10,15,6,12,46,tzinfo=datetime.UTC).timestamp()
while not source.exists():
    if time.time()>deadline:raise RuntimeError('Native comparison did not finish in the study budget')
    time.sleep(30)
if json.loads(source.read_text())['status']!='COMPLETE':raise RuntimeError('Native comparison failed')
argv=['taskset','-c','14',str(root/'genesis_tools/.venv/bin/python'),str(root/'benchmarks/experiments/plant-methods-20261013/compare_production.py'),'--native',str(source.parent),'--workflow-scores',str(base/'whole-workflow-comparison'),'--genesis-run','/var/tmp/genesis-public-workflow-20261013/genesis-results','--output',str(root/'validation/evidence/method-comparison-20261013/atac-native-production.json')]
start=time.monotonic();utc=datetime.datetime.now(datetime.UTC).isoformat()
r=subprocess.run(argv,env=dict(os.environ,PYTHONPATH=str(root/'genesis_tools/src')))
(base/'native-production-command.json').write_text(json.dumps({'argv':argv,'utc':utc,'exit':r.returncode,'seconds':time.monotonic()-start},indent=2)+'\n')
raise SystemExit(r.returncode)
